# Render blueprint: one Docker web service, one port, WebSockets supported.
# Deploy: push this repo, then "New > Blueprint" in Render.
services:
  - type: web
    name: inolas-messenger
    runtime: docker
    dockerfilePath: ./Dockerfile
    dockerCommand: ./start.sh
    plan: starter
    healthCheckPath: /
    autoDeploy: true
    envVars:
      # Render sets PORT automatically; Caddy binds to it.
      - key: BACKEND_PORT
        value: "8000"
      # Rebuild the frontend at boot so API_URL matches the Render domain.
      - key: REFLEX_EXPORT_ON_START
        value: "1"
      - key: API_URL
        sync: false # e.g. https://inolas-messenger.onrender.com
      - key: APP_BASE_URL
        sync: false # same value; used for the OAuth callback URL
      - key: SUPABASE_URL
        sync: false
      - key: SUPABASE_KEY
        sync: false
      - key: OPENAI_API_KEY
        sync: false
      - key: REFLEX_DB_URL
        sync: false
      - key: GOOGLE_CLIENT_ID
        sync: false
      - key: GOOGLE_CLIENT_SECRET
        sync: false
